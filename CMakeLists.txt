cmake_minimum_required(VERSION 3.20)
project(StarRunner VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(CTest)
add_executable(game_tests tests/game_test.cpp)
target_include_directories(game_tests PRIVATE src)
add_test(NAME gameplay COMMAND game_tests)
option(BUILD_GAME "Build the graphical game" ON)
if(BUILD_GAME)
    include(FetchContent)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(raylib)
    add_executable(StarRunner src/main.cpp)
    target_link_libraries(StarRunner PRIVATE raylib)
    if(WIN32)
        set_target_properties(StarRunner PROPERTIES WIN32_EXECUTABLE TRUE)
        if(MSVC)
            target_link_options(StarRunner PRIVATE /ENTRY:mainCRTStartup)
        endif()
    endif()
    install(TARGETS StarRunner RUNTIME DESTINATION .)
    install(FILES README.md DESTINATION .)
endif()
